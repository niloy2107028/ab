!4. Router0 — IPsec — Where to paste: Router0 → CLI
!configure terminal
!
!crypto isakmp policy 10
!encryption aes
!authentication pre-share
!group 2
!exit
!
!crypto isakmp key cisco123 address 20.0.0.1
!
!crypto ipsec transform-set SET esp-aes esp-sha-hmac
!
!access-list 110 permit gre host 10.0.0.1 host 20.0.0.1
!
!crypto map MAP 10 ipsec-isakmp
!set peer 20.0.0.1
!set transform-set SET
!match address 110
!exit
!
!interface FastEthernet0/0
!crypto map MAP
!exit
!
!end
!5. Router2 — IPsec — Where to paste: Router2 → CLI
!configure terminal

!group 2
!exit
!
!crypto isakmp key cisco123 address 10.0.0.1
!
!crypto ipsec transform-set SET esp-aes esp-sha-hmac
!
!access-list 110 permit gre host 20.0.0.1 host 10.0.0.1
!
!crypto map MAP 10 ipsec-isakmp
!set peer 10.0.0.1
!set transform-set SET
!match address 110
!exit
!
!interface FastEthernet0/0
!crypto map MAP
!exit
!
!end
